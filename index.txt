async function sbPost(table, data) {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/${table}`, {
    method: 'POST',
    headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}`, 'Content-Type': 'application/json', 'Prefer': 'return=representation' },
    body: JSON.stringify(data)
  });
  return res.json();
}

    body: JSON.stringify(data)
  });
  return res.json();
}

function soon(txt) {
  if (!txt) return false;
  const [d,m,y] = txt.split('/').map(Number);
  if (!d||!m||!y) return false;
  const dt = new Date(y, m-1, d);
  return ((dt - new Date()) / (1000*60*60*24)) <= 10;
}

function autoSuggestTrack() {
  const systemEl = document.getElementById('f_system');
  const trackEl = document.getElementById('f_track_type');
  const sizeEl = document.getElementById('f_system_size_kw');
  if (!systemEl || !trackEl) return;
  
  const system = systemEl.value;
  const currentTrack = trackEl.value;
  const size = Number(sizeEl?.value) || 0;
  
  let suggestedTrack = currentTrack;
  
  if (system === 'OFF GRID') {
    suggestedTrack = 'off_grid';
  } else if (system === 'ON GRID' || system === 'HYBRID') {
    if (currentTrack === 'off_grid') {
      suggestedTrack = (size > 0 && size <= 15) ? 'green_track' : 'regular';
    }
  }
  
  if (suggestedTrack !== currentTrack) {
    trackEl.value = suggestedTrack;
    trackEl.style.background = '#fef3c7';
    setTimeout(() => { trackEl.style.background = ''; }, 2000);
  }
}

async function saveClient() {
  const name = document.getElementById('f_name').value.trim();
  if (!name) { alert('יש להזין שם לקוח'); return; }
  
  setSyncStatus('saving');
  
  const papersVal = document.getElementById('f_progress_papers').value;
  const papers = papersVal === '' ? null : Number(papersVal);
  const equipment = Number(document.getElementById('f_progress_equipment').value) || 0;
  const installation = Number(document.getElementById('f_progress_installation').value) || 0;
  const final = Number(document.getElementById('f_progress_final').value) || 0;
  const overall = Math.round(((papers || 0) + equipment + installation + final) / 4 * 10) / 10;
  
  const clientData = {
    name,
    email: document.getElementById('f_email').value,
    phone: document.getElementById('f_phone').value,
    location: document.getElementById('f_location').value,
    system: document.getElementById('f_system').value,
    track_type: document.getElementById('f_track_type')?.value || 'regular',
    system_size_kw: Number(document.getElementById('f_system_size_kw')?.value) || null,
    site_surveyor: document.getElementById('f_site_surveyor')?.value || null,
    customer_type: document.getElementById('f_customer_type').value,
    zoho_amount: Number(document.getElementById('f_zoho_amount').value) || null,
    progress_papers: papers,
    progress_equipment: equipment,
    progress_installation: installation,
    progress_final: final,
    progress_overall: overall,
    risk: document.getElementById('f_risk').value,
    blocker: document.getElementById('f_blocker').value,
    next_action: document.getElementById('f_next_action').value,
    deadline: document.getElementById('f_deadline').value,
    deliveries: document.getElementById('f_deliveries').value.split(',').map(s=>s.trim()).filter(Boolean)
  };
  
  try {
    let clientId;
    let trackChanged = false;
    
    if (editingClient) {
      trackChanged = editingClient.track_type !== clientData.track_type;
      await sbPatch('clients', editingClient.id, clientData);
      clientId = editingClient.id;
      
      if (trackChanged) {
        const oldTrackLabel = trackLabel(editingClient.track_type || 'regular');
        const newTrackLabel = trackLabel(clientData.track_type);
        const shouldReset = confirm(
          `🛤️ שינית את המסלול מ: ${oldTrackLabel} ל: ${newTrackLabel}\n\n` +
          `האם למשוך את השלבים המפורטים של המסלול החדש?\n\n` +
          `(שים לב: השלבים הקודמים יתאפסו. פרטי הלקוח ישמרו.)`
        );
        
        if (shouldReset) {
          await sbDelete('client_stages', `client_id=eq.${clientId}`);
          await sbDelete('client_parallel', `client_id=eq.${clientId}`);
          
          const newStages = workflowTemplates.filter(t => t.track_type === clientData.track_type);
          
          if (newStages.length > 0) {
            await sbPost('client_stages', newStages.filter(s => !s.is_parallel).map(s => ({
              client_id: clientId, text: s.stage_name, status: 'waiting', order_index: s.stage_number
            })));
            
            const parallelStages = newStages.filter(s => s.is_parallel);
            if (parallelStages.length > 0) {
              await sbPost('client_parallel', parallelStages.map(s => ({
                client_id: clientId, text: s.stage_name, status: 'parallel', order_index: s.stage_number
              })));
            }
          }
          closeEditModal();
          await loadAll();
          showToast(`השלבים עודכנו למסלול ${newTrackLabel}!`, 'success');
          return;
        }
      }
    } else {
      const result = await sbPost('clients', clientData);
      clientId = result[0].id;
      
      // Load stages from template automatically for new clients
      const newStages = workflowTemplates.filter(t => t.track_type === clientData.track_type);
      if (newStages.length > 0) {
        await sbPost('client_stages', newStages.filter(s => !s.is_parallel).map(s => ({
          client_id: clientId, text: s.stage_name, status: 'waiting', order_index: s.stage_number
        })));
      }
      closeEditModal();
      await loadAll();
      showToast('לקוח חדש נוסף בהצלחה', 'success');
      return;
    }
    
    // Save manual modifications to stages
    if (tempStages.length > 0) {
      await sbDelete('client_stages', `client_id=eq.${clientId}`);
      await sbPost('client_stages', tempStages.filter(s=>s.text).map((s, i) => ({
        client_id: clientId, text: s.text, status: s.status, order_index: i
      })));
    }
    
    closeEditModal();
    await loadAll();
    showToast('השינויים נשמרו', 'success');
  } catch (err) {
    setSyncStatus('error');
    alert('שגיאה בשמירה: ' + err.message);
  }
}

async function doLogin() {
  const email = document.getElementById('loginEmail').value.trim();
  const password = document.getElementById('loginPassword').value;
  const errorEl = document.getElementById('loginError');
  const btn = document.getElementById('loginBtn');
  
  errorEl.classList.add('hidden');
  
  if (!email || !password) {
    errorEl.textContent = 'יש למלא מייל וסיסמה';
    errorEl.classList.remove('hidden');
    return;
  }
  
  btn.disabled = true;
  btn.textContent = '⏳ מתחבר...';
  
  try {
    const res = await fetch(`${SUPABASE_URL}/auth/v1/token?grant_type=password`, {
      method: 'POST',
      headers: { 'apikey': SUPABASE_KEY, 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    const data = await res.json();
    
    if (data.error || data.error_description || !data.access_token) {
      errorEl.textContent = data.error_description || data.msg || 'שגיאה בהתחברות';
      errorEl.classList.remove('hidden');
      btn.disabled = false;
      btn.textContent = '🔐 התחבר';
      return;
    }
    
    accessToken = data.access_token;
    currentUser = data.user;
    localStorage.setItem('doxy_session', JSON.stringify({
      access_token: data.access_token,
      refresh_token: data.refresh_token,
      expires_at: data.expires_at,
      user: data.user
    }));
    
    showApp();
  } catch (err) {
    errorEl.textContent = 'שגיאה: ' + err.message;
    errorEl.classList.remove('hidden');
    btn.disabled = false;
    btn.textContent = '🔐 התחבר';
  }
}

async function checkAuth() {
  const saved = localStorage.getItem('doxy_session');
  if (saved) {
    try {
      const session = JSON.parse(saved);
      if (session.expires_at && session.expires_at * 1000 > Date.now()) {
        accessToken = session.access_token;
        currentUser = session.user;
        showApp();
        return true;
      } else if (session.refresh_token) {
        const refreshed = await refreshSession(session.refresh_token);
        if (refreshed) return true;
      }
    } catch {}
  }
  showLogin();
  return false;
}
async function refreshSession(refreshToken) {
  try {
    const res = await fetch(`${SUPABASE_URL}/auth/v1/token?grant_type=refresh_token`, {
      method: 'POST',
      headers: { 'apikey': SUPABASE_KEY, 'Content-Type': 'application/json' },
      body: JSON.stringify({ refresh_token: refreshToken })
    });
    const data = await res.json();
    if (data.access_token) {
      accessToken = data.access_token;
      currentUser = data.user;
      localStorage.setItem('doxy_session', JSON.stringify({
        access_token: data.access_token,
        refresh_token: data.refresh_token,
        expires_at: data.expires_at,
        user: data.user
      }));
      showApp();
      return true;
    }
  } catch {}
  return false;
}

window.fetch = function(url, options = {}) {
  if (typeof url === 'string' && url.startsWith(SUPABASE_URL) && accessToken) {
    options.headers = options.headers || {};
    if (url.includes('/rest/v1/') || url.includes('/storage/v1/') || url.includes('/functions/v1/')) {
      options.headers['Authorization'] = `Bearer ${accessToken}`;
    }
  }
  return originalFetch.call(this, url, options);
};
